local function formatNumber(num)
    local n = tonumber(num) or 0
    local abs = math.abs(n)
    if abs >= 1e12 then return string.format("%.2fT", n / 1e12) end
    if abs >= 1e9 then return string.format("%.2fB", n / 1e9) end
    if abs >= 1e6 then return string.format("%.2fm", n / 1e6) end
    if abs >= 1e3 then return string.format("%.1fk", n / 1e3) end
    return string.format("%.0f", n)
end

local function getRarityColor(n)
    n = tonumber(n) or 0
    if n >= 11 then return Color3.fromRGB(255,236,150) end
    if n >= 10 then return Color3.fromRGB(255,90,210) end
    if n >= 9 then return Color3.fromRGB(120,210,255) end
    if n >= 8 then return Color3.fromRGB(255,80,110) end
    if n >= 7 then return Color3.fromRGB(255,186,70) end
    if n >= 6 then return Color3.fromRGB(186,120,255) end
    if n >= 5 then return Color3.fromRGB(255,220,90) end
    if n >= 4 then return Color3.fromRGB(160,120,255) end
    if n >= 3 then return Color3.fromRGB(90,170,255) end
    if n >= 2 then return Color3.fromRGB(120,200,120) end
    return Color3.fromRGB(210,210,210)
end

local function getEggPosition(egg)
    if not egg then return nil end
    local pos = egg.BottomCFrame
    if pos then
        if typeof(pos) == "Vector3" then return pos + Vector3.new(0,4,0) end
        if typeof(pos) == "CFrame" then return pos.Position + Vector3.new(0,4,0) end
    end
    pos = egg.BoundsCFrame
    if pos then
        if typeof(pos) == "Vector3" then return pos end
        if typeof(pos) == "CFrame" then return pos.Position end
    end
    pos = egg.CFrame
    if pos then
        if typeof(pos) == "Vector3" then return pos end
        if typeof(pos) == "CFrame" then return pos.Position end
    end
    pos = egg.Position
    if pos then
        if typeof(pos) == "Vector3" then return pos + Vector3.new(4,0,4) end
        if typeof(pos) == "CFrame" then return pos.Position + Vector3.new(4,0,4) end
    end
    return nil
end

local function getEggEarn(egg, cfg)
    if type(egg) ~= "table" then return 0 end
    if not cfg and egg then
        local cat = egg.AssetCategory or egg.Category
        if Directory and cat then cfg = Directory[cat] end
    end
    local num = tonumber(egg.MoneyPerSecond)
    if num and num > 0 then return num end
    local baseRate = type(cfg) == "table" and (tonumber(cfg.EarningRate) or 0) or 0
    local mutations = {}
    if type(egg.Mutations) == "table" then mutations = egg.Mutations end
    local scale = tonumber(egg.Scale) or 1
    if scale <= 0 then scale = 1 end
    local weight = tonumber(egg.Weight) or tonumber(egg.Kg) or tonumber(egg.ModelWeight) or 0
    local modelWeight = type(cfg) == "table" and tonumber(cfg.ModelWeight) or 0
    if weight > 0 and modelWeight > 0 then scale = (weight / modelWeight) ^ (1/3) end
    local scaledEarn = scale <= 5 and scale ^ 1.85 or (scale / 5) ^ 1.2 * 19.637875755794
    local mutMult = 1
    pcall(function()
        local Mutations = require(ReplicatedStorage.Shared.Modules.Mutations)
        if Mutations and Mutations.EarningsFor then mutMult = Mutations.EarningsFor(mutations) or 1 end
    end)
    local result = baseRate * scaledEarn * mutMult
    if result < 1 and baseRate > 0 then result = 1
    return math.floor(result + 0.5)
end

function ESP:cleanPool()
    for key, item in pairs(self.pool) do
        if not item.alive then
            if item.bb then item.bb:Destroy() end
            if item.dummy then pcall(function() item.dummy:Destroy() end) end
            self.pool[key] = nil
        else
            item.alive = false
        end
    end
end

function ESP:getEggDisplay(egg, cfg, isTarget)
    local name = cfg and (cfg.DisplayName or (type(cfg.Egg) == "table" and cfg.Egg.DisplayName)) or tostring(egg.AssetCategory or "?")
    if isTarget then name = "▸ " .. tostring(name) end
    local earn = formatNumber(getEggEarn(egg, cfg)) .. "/s"
    local rarity = type(cfg) == "table" and type(cfg.Rarity) == "table" and tostring(cfg.Rarity.DisplayName or cfg.Rarity.Name or cfg.Rarity._id or "?") or "?"
    local weight = ""
    local w = 0
    if type(egg) == "table" then w = tonumber(egg.Weight) or tonumber(egg.ModelWeight) or tonumber(egg.Kg) or 0 end
    if w <= 0 and type(cfg) == "table" then w = tonumber(cfg.Weight) or tonumber(cfg.ModelWeight) or 0 end
    if w > 0 then weight = w >= 100 and string.format("%.0fkg", w) or string.format("%.1fkg", w) end
    local chance = ""
    if type(cfg) == "table" and type(cfg.Rarity) == "table" then
        local val = cfg.Rarity.DefaultRarityValue
        if type(val) == "string" and val ~= "" then
            local cleaned = string.gsub(val, ",", "")
            cleaned = string.gsub(cleaned, "1 in ", "1/")
            cleaned = string.gsub(cleaned, " ", "")
            if string.sub(cleaned, 1, 2) == "1/" then
                local num = tonumber(string.sub(cleaned, 3))
                if num then
                    if num >= 1e12 then chance = string.format("1/%.2ft", num / 1e12)
                    elseif num >= 1e9 then chance = string.format("1/%.2fb", num / 1e9)
                    elseif num >= 1e6 then chance = string.format("1/%.2fm", num / 1e6)
                    elseif num >= 1e3 then chance = string.format("1/%.0fk", num / 1e3)
                    else chance = cleaned end
                end
            end
        end
    end
    local parts = {earn, rarity}
    if chance ~= "" then table.insert(parts, chance) end
    if weight ~= "" then table.insert(parts, weight) end
    local mutations = ""
    if type(egg) == "table" and type(egg.Mutations) == "table" and #egg.Mutations > 0 then mutations = table.concat(egg.Mutations, " · ") end
    local sub = table.concat(parts, " · ")
    if mutations ~= "" then sub = sub .. "\n" .. mutations end
    local rarityNum = type(cfg) == "table" and type(cfg.Rarity) == "table" and tonumber(cfg.Rarity.RarityNumber) or 0
    local color = getRarityColor(rarityNum)
    local icon = nil
    local cat = egg and egg.AssetCategory
    if cat and cfg then
        local keys = {}
        local function addKey(k) if type(k) == "string" and k ~= "" then local lk = string.lower(k) table.insert(keys, lk) table.insert(keys, lk:gsub("[%s_%-]+","")) end end
        addKey(cat)
        addKey(cfg.DisplayName)
        addKey(cfg._id)
        if type(cfg.Egg) == "table" then addKey(cfg.Egg.DisplayName) end
        for _, k in ipairs(keys) do
            local ic = self.iconBy[k]
            if ic then icon = ic break end
        end
    end
    return {name = name, sub = sub, color = color, target = isTarget, tall = mutations ~= "", icon = icon}
end

function ESP:readFieldEggs()
    if not u1103 then return {} end
    local readFunc = u1103.ReadFieldEggs or u1103.SyncFieldEggs
    if type(readFunc) ~= "function" then return {} end
    local ok, result = pcall(readFunc)
    if not ok or type(result) ~= "table" or type(result.Records) ~= "table" then return {} end
    local eggs = {}
    local seen = {}
    for _, v in pairs(result.Records) do
        if type(v) == "table" and v.Uid and not seen[v.Uid] then
            seen[v.Uid] = true
            table.insert(eggs, v)
        end
    end
    return eggs
end

function ESP:isPlotEgg(egg)
    if not egg then return false end
    if egg.Placement ~= nil then return true end
    if egg.OwnerUserId and egg.State ~= "Dropped" then return true end
    local uid = string.lower(tostring(egg.Uid or ""))
    if uid:find("plot", 1, true) or uid:find(":pen", 1, true) or uid:find("hatch", 1, true) then return true end
    return false
end

function ESP:readPlayerEggs()
    if not u1103 then return nil end
    if type(u1103.ReadOwnerEggs) == "function" then
        local ok, result = pcall(u1103.ReadOwnerEggs, LocalPlayer.UserId)
        if ok and type(result) == "table" then return result end
    end
    if type(u1103.ReadOwnedEggs) == "function" then
        local ok, result = pcall(u1103.ReadOwnedEggs)
        if ok and type(result) == "table" then
            for _, v in pairs(result) do
                if type(v) == "table" and tonumber(v.OwnerUserId) == LocalPlayer.UserId then
                    return v.Records
                end
            end
        end
    end
    return nil
end

function ESP:getPlot()
    local plot = nil
    pcall(function() plot = require(ReplicatedStorage.Client.PlotState).ResolvePlot() end)
    if type(plot) == "table" and plot.PlotFolder then return plot.PlotFolder, plot.CenterPoint end
    local Plots = workspace:FindFirstChild("Plots")
    if not Plots then return end
    for _, child in ipairs(Plots:GetChildren()) do
        local found = false
        for _, desc in ipairs(child:GetDescendants()) do
            if desc:IsA("TextLabel") and string.find(desc.Text, LocalPlayer.Name, 1, true) then
                found = true
                break
            end
        end
        if found then
            local center = child:FindFirstChild("CenterPoint", true)
            return child, center
        end
    end
    return nil, nil
end

function ESP:isEggReady(egg, uid)
    uid = uid or egg.Uid
    if uid and u1103 and type(u1103.IsReadyToHatch) == "function" then
        local ok, result = pcall(u1103.IsReadyToHatch, uid)
        if ok and result then return 0, true end
    end
    if egg.IsReady == true or egg.Ready == true then return 0, true end
    local endsAt = egg.HatchEndsAt or egg.ReadyAt or egg.GrowEndsAt
    if type(endsAt) == "number" then
        local left = endsAt - workspace.DistributedGameTime
        return left, left <= 0
    end
    local left = tonumber(egg.TimeLeft) or tonumber(egg.HatchTimeLeft)
    if left then return left, left <= 0 end
    return nil, false
end

function ESP:formatTime(seconds)
    local s = math.max(0, math.floor(tonumber(seconds) or 0))
    local h = math.floor(s / 3600)
    local m = math.floor(s % 3600 / 60)
    local sec = s % 60
    if h > 0 then return string.format("%d:%02d:%02d", h, m, sec) end
    return string.format("%d:%02d", m, sec)
end

function ESP:readPlotEggs()
    local elapsed = os.clock()
    if elapsed - (self.plotAt or 0) < 0.5 then return self.plotList end
    self.plotAt = elapsed
    local list = {}
    local eggs = self:readPlayerEggs()
    local plotFolder, centerPoint = self:getPlot()
    local cframe = CFrame.new()
    if centerPoint and centerPoint:IsA("BasePart") then cframe = centerPoint.CFrame
    elseif plotFolder then pcall(function() cframe = plotFolder:GetPivot() end) end
    if type(eggs) == "table" then
        for k, v in pairs(eggs) do
            if type(v) == "table" and v.AssetCategory then
                local uid = v.Uid or k
                local cat = v.AssetCategory
                local cfg = Directory and Directory[cat]
                local pos = nil
                local placement = v.Placement
                if type(placement) == "table" and placement.LocalCFrame then
                    local ok, result = pcall(function() return cframe * placement.LocalCFrame end)
                    if ok and result then pos = result.Position + Vector3.new(0,3,0) end
                end
                if pos then
                    local left, ready = self:isEggReady(v, uid)
                    table.insert(list, {key = "p:" .. tostring(uid), pos = pos, cfg = cfg, rec = v, left = left, ready = ready})
                end
            end
        end
    end
    self.plotList = list
    return list
end

function ESP:tick()
    if not (CONFIG.EggESP or CONFIG.PlotESP or CONFIG.ESPBeam) then return end
    local elapsed = os.clock()
    if elapsed - (self.lastTick or 0) < 0.08 then return end
    self.lastTick = elapsed
    if CONFIG.EggESP then
        local eggs = self:readFieldEggs()
        for _, egg in ipairs(eggs) do
            if type(egg) ~= "table" or not egg.Uid then continue end
            local state = egg.State
            if state ~= "Slot" and state ~= "Dropped" then continue end
            if self:isPlotEgg(egg) then continue end
            local cat = egg.AssetCategory
            local cfg = Directory and Directory[cat]
            local pos = getEggPosition(egg)
            if not pos then
                local poolItem = self.pool["f:" .. egg.Uid]
                pos = poolItem and poolItem.pos
            end
            if pos then
                local data = self:getEggDisplay(egg, cfg, false)
                self:updateBillboard("f:" .. egg.Uid, pos, data)
            end
        end
    end
    if CONFIG.PlotESP then
        for _, egg in ipairs(self:readPlotEggs()) do
            local cfg = egg.cfg
            local earn = formatNumber(getEggEarn(egg.rec, cfg)) .. "/s"
            local name = cfg and (cfg.DisplayName or (egg.rec and egg.rec.AssetCategory)) or "plot egg"
            local status = egg.ready and "ready" or (egg.left and self:formatTime(egg.left) or "hatching")
            local sub = earn .. "  ·  " .. status
            local data = {name = tostring(name), sub = sub, color = egg.ready and Color3.fromRGB(138,166,128) or Color3.fromRGB(168,158,144), target = egg.ready == true}
            local cat = egg.rec and egg.rec.AssetCategory
            if cat and cfg then
                local keys = {}
                local function addKey(k) if type(k) == "string" and k ~= "" then local lk = string.lower(k) table.insert(keys, lk) table.insert(keys, lk:gsub("[%s_%-]+","")) end end
                addKey(cat)
                addKey(cfg.DisplayName)
                addKey(cfg._id)
                if type(cfg.Egg) == "table" then addKey(cfg.Egg.DisplayName) end
                for _, k in ipairs(keys) do
                    local ic = self.iconBy[k]
                    if ic then data.icon = ic break end
                end
            end
            self:updateBillboard(egg.key, egg.pos, data)
        end
    end
    if CONFIG.ESPBeam and _G.t216 and _G.t216.running and _G.t216.target and _G.t216.target.pos then
        local char = LocalPlayer.Character
        if char then
            local root = char:FindFirstChild("HumanoidRootPart")
            if root then self:updateBeam(root, _G.t216.target.pos) end
        end
    elseif self.beamObj then
        self.beamObj.Enabled = false
    end
    self:cleanPool()
end

function ESP:updateStats()
    if not CONFIG.StatsPanel then
        if self.statsFrame then self.statsFrame.Visible = false end
        return
    end
    local elapsed = os.clock()
    if elapsed - (self.statsAt or 0) < 0.35 then return end
    self.statsAt = elapsed
    local frame = self:createStatsPanel()
    if not frame then return end
    frame.Visible = true
    local rows = self.statsRows
    if not rows then return end
    local sessionTime = math.max(0, math.floor(elapsed - self.sessionAt))
    local h = math.floor(sessionTime / 3600)
    local m = math.floor(sessionTime % 3600 / 60)
    local s = sessionTime % 60
    local sessionStr = (h > 0 and string.format("%d:%02d:%02d", h, m, s) or string.format("%d:%02d", m, s))
    if rows.Session and rows.Session.Parent then rows.Session.Text = sessionStr end
    pcall(function()
        local money = "—"
        local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
        if leaderstats then
            local cash = leaderstats:FindFirstChild("Money") or leaderstats:FindFirstChild("Cash") or leaderstats:FindFirstChild("Coins")
            if cash and cash:IsA("ValueBase") then money = "$" .. formatNumber(cash.Value) end
        end
        if rows.Money and rows.Money.Parent then rows.Money.Text = money end
    end)
    pcall(function()
        local income = "—"
        local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
        if leaderstats then
            local inc = leaderstats:FindFirstChild("Money/s") or leaderstats:FindFirstChild("Income")
            if inc and inc:IsA("ValueBase") then income = formatNumber(inc.Value) .. "/s" end
        end
        if rows.Income and rows.Income.Parent then rows.Income.Text = income end
    end)
    pcall(function()
        local speed = "—"
        local leaderstats = LocalPlayer:FindFirstChild("leaderstats")
        if leaderstats then
            local spd = leaderstats:FindFirstChild("Speed")
            if spd and spd:IsA("ValueBase") then speed = formatNumber(spd.Value) end
        end
        if rows.Speed and rows.Speed.Parent then rows.Speed.Text = speed end
    end)
    pcall(function()
        local placed = 0
        local eggs = self:readPlayerEggs()
        if type(eggs) == "table" then
            for _, v in pairs(eggs) do
                if type(v) == "table" and v.Placement then placed = placed + 1 end
            end
        end
        if rows.Pen and rows.Pen.Parent then rows.Pen.Text = tostring(placed) .. " placed" end
    end)
    pcall(function()
        local best = "—"
        local bestEarn = 0
        local eggs = self:readPlayerEggs()
        if type(eggs) == "table" then
            for _, v in pairs(eggs) do
                if type(v) == "table" then
                    local cat = v.AssetCategory
                    local cfg = Directory and Directory[cat]
                    local earn = getEggEarn(v, cfg)
                    if earn > bestEarn then
                        bestEarn = earn
                        best = cfg and (cfg.DisplayName or type(cfg.Egg) == "table" and cfg.Egg.DisplayName) or cat or "?"
                    end
                end
            end
        end
        if best ~= "—" then best = best .. "\n" .. formatNumber(bestEarn) .. "/s" end
        if rows["Best pet"] and rows["Best pet"].Parent then rows["Best pet"].Text = best end
    end)
    pcall(function()
        local best = "—"
        local bestEarn = 0
        local eggs = self:readFieldEggs()
        if type(eggs) == "table" then
            for _, v in ipairs(eggs) do
                if type(v) == "table" and (v.State == "Slot" or v.State == "Dropped") and not self:isPlotEgg(v) then
                    local cat = v.AssetCategory
                    local cfg = Directory and Directory[cat]
                    local earn = getEggEarn(v, cfg)
                    if earn > bestEarn then
                        bestEarn = earn
                        best = cfg and (cfg.DisplayName or type(cfg.Egg) == "table" and cfg.Egg.DisplayName) or cat or "?"
                    end
                end
            end
        end
        if best ~= "—" then best = best .. "\n" .. formatNumber(bestEarn) .. "/s" end
        if rows["Best egg"] and rows["Best egg"].Parent then rows["Best egg"].Text = best end
    end)
end
